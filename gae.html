  <script>

    (function(){

      const btn = document.getElementById('revealBtn');

      const img = document.getElementById('meme');


      let revealed = false;


      btn.addEventListener('click', function(){

        // Prevent double clicks

        if (revealed) return;


        revealed = true;

        btn.setAttribute('aria-expanded', 'true');

        // add a class to the <html> element so CSS can change the background

        document.documentElement.classList.add('revealed');


        // show the image by toggling its opacity via CSS (already prepared)

        // ensure the image's parent is no longer aria-hidden for accessibility

        img.parentElement.setAttribute('aria-hidden', 'false');


        // optional: scroll image into view on small screens

        setTimeout(() => {

          img.scrollIntoView({behavior:'smooth', block:'center'});

        }, 120);


        // disable the button to indicate it's been used

        btn.disabled = true;

      });


      // graceful handling if image fails to load

      img.addEventListener('error', () => {

        img.alt = 'Image failed to load. Make sure the file name is correct and the image is in the same folder.';

        img.style.display = 'none';

      });

    })();

  </script>
